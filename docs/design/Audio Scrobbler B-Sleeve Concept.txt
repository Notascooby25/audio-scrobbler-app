      nextLikedState = !existingTrack.liked;
      setTracks((prev) =>
        prev.map((t) => (t.id === trackId ? { ...t, liked: nextLikedState } : t))
      );

      prevGigs.map((gig) => ({

        setlist: gig.setlist.map((s) => {
          if (s.id === trackId || s.title === title) {
            nextLikedState = existingTrack ? nextLikedState : !s.liked;
            return { ...s, liked: nextLikedState };
          }
          return s;
        })
      }))
    );

                  : 'border border-white/25 hover:border-white'
              }`}
              aria-label="Open profile"
            >
              CL
            </button>
          </div>
        )}
      </header>
    );
  };

  // Bottom Navigation Bar (Matching Mockup: Today | Library | Play | Reports | Gigs)
  const renderBottomNav = () => {
    if (activeScreen === 'first-run') return null;

    const navItems = [
      { id: 'today', label: 'Today', icon: Activity },
      { id: 'library', label: 'Library', icon: Library },
      {
        id: 'play-toggle',
        label: 'Play',
        icon: Play,
        onClick: () => {
          setIsVinylSpinning(!isVinylSpinning);
          setActiveScreen('today');
          triggerToast(
            !isVinylSpinning ? 'Resumed sleeve playback' : 'Paused vinyl sleeve animation'
          );
        }
      },
      { id: 'reports', label: 'Reports', icon: BarChart2 },
      { id: 'gigs', label: 'Gigs', icon: Ticket }
    ];

    return (
      <nav
        style={{ backgroundColor: 'rgba(0, 0, 0, 0.22)' }}
        className="border-t border-white/10 px-3 py-2 flex items-center justify-around backdrop-blur-md"
      >
        {navItems.map((item) => {
          const IconComponent = item.icon;
          const isSelected =
            activeScreen === item.id ||
            (item.id === 'gigs' && activeScreen === 'gig-detail');

          return (
            <button
              key={item.id}
              type="button"
              onClick={() => (item.onClick ? item.onClick() : setActiveScreen(item.id))}
              className="flex flex-col items-center justify-center gap-1 min-w-[54px] py-1 group focus:outline-none"
            >
              <div
                style={
                  isSelected
                    ? { backgroundColor: activeTheme.navPillBg || 'rgba(255,255,255,0.2)' }
                    : undefined
                }
                className={`px-3.5 py-1 rounded-full transition-colors flex items-center justify-center ${
                  isSelected
                    ? 'text-white border border-white/15'
                    : 'text-white/65 group-hover:text-white'
                }`}
              >
                <IconComponent size={16} />
              </div>
              <span
                className={`text-[10px] font-medium ${
                  isSelected ? 'text-white font-semibold' : 'text-white/65'
                }`}
              >
                {item.label}
              </span>
            </button>
          );
        })}
      </nav>
    );
  };

  const ALL_MOCKUP_SCREENS = [
    { id: 'today', label: '1. Today (Vinyl)' },
    { id: 'library', label: '2. Library (♥ Likes)' },
    { id: 'gigs', label: '3. Gigs' },
    { id: 'gig-detail', label: '4. Gig Detail' },
    { id: 'connect', label: '5. Connect' },
    { id: 'profile', label: '6. Profile' },
    { id: 'following', label: '7. Following' },
    { id: 'settings', label: '8. Settings' },
    { id: 'first-run', label: '9. First Run' }
  ];

  return (
    <div className="min-h-screen bg-[#111114] text-zinc-100 flex flex-col font-sans selection:bg-rose-500 selection:text-white">
      {/* TOP CONTROL & VIEWPORT BAR */}
      <div className="sticky top-0 z-40 bg-[#18181c]/95 backdrop-blur-md border-b border-white/10 px-4 py-2.5">
        <div className="max-w-7xl mx-auto flex flex-col lg:flex-row lg:items-center lg:justify-between gap-3">
          {/* Brand & Active Sleeve Theme Indicator */}
          <div className="flex items-center justify-between gap-4">
            <div className="flex items-center gap-2.5">
              <div
                style={{ backgroundColor: activeTheme.heroCard }}
                className="w-7 h-7 rounded-md flex items-center justify-center text-white font-serif font-bold text-sm shadow-inner border border-white/20 transition-colors duration-500"
              >
                B
              </div>
              <div>
                <div className="flex items-center gap-2">
                  <span className="text-sm font-bold tracking-tight text-white">
                    B — Sleeve Concept Preview
                  </span>
                  <span className="text-[11px] px-2 py-0.5 rounded-full bg-white/10 text-white/80 font-medium">
                    {activeTheme.name}
                  </span>
                </div>
                <p className="text-[11px] text-zinc-400 hidden sm:block">
                  Notascooby25/audio-scrobbler-app • Interactive Mobile PWA & Desktop Mode
                </p>
              </div>
            </div>

            {/* Mobile/Desktop Viewport Toggle */}
            <div className="flex items-center bg-black/50 p-1 rounded-lg border border-white/10">
              <button
                type="button"
                onClick={() => setViewportMode('mobile')}
                className={`px-3 py-1.5 rounded-md text-xs font-semibold inline-flex items-center gap-1.5 transition ${
                  viewportMode === 'mobile'
                    ? 'bg-white text-zinc-900 shadow-sm'
                    : 'text-zinc-400 hover:text-white'
                }`}
              >
                <Smartphone size={14} />
                Mobile (390px)
              </button>
              <button
                type="button"
                onClick={() => setViewportMode('desktop')}
                className={`px-3 py-1.5 rounded-md text-xs font-semibold inline-flex items-center gap-1.5 transition ${
                  viewportMode === 'desktop'
                    ? 'bg-white text-zinc-900 shadow-sm'
                    : 'text-zinc-400 hover:text-white'
                }`}
              >
                <Monitor size={14} />
                Desktop Mode
              </button>
            </div>
          </div>

          {/* Screen Switcher Pills & 44px Hitbox Inspector Toggle */}
          <div className="flex items-center gap-1.5 overflow-x-auto pb-1 lg:pb-0 no-scrollbar">
            {ALL_MOCKUP_SCREENS.map((scr) => (
              <button
                key={scr.id}
                type="button"
                onClick={() => setActiveScreen(scr.id)}
                className={`px-2.5 py-1 rounded-full text-xs font-medium whitespace-nowrap transition ${
                  activeScreen === scr.id
                    ? 'bg-rose-600 text-white shadow-sm'
                    : 'bg-white/5 text-zinc-300 hover:bg-white/10'
                }`}
              >
                {scr.label}
              </button>
            ))}

            <button
              type="button"
              onClick={() => setShowHitboxGuides(!showHitboxGuides)}
              className={`ml-1 px-2.5 py-1 rounded-full text-xs font-medium whitespace-nowrap inline-flex items-center gap-1 border transition ${
                showHitboxGuides
                  ? 'bg-rose-500/20 border-rose-400 text-rose-200'
                  : 'border-white/15 text-zinc-400 hover:text-white'
              }`}
              title="Highlight 44×44px minimum touch target hitboxes on Heart buttons"
            >
              <Heart size={11} className="fill-rose-400 text-rose-400" />
              44px Hitbox
            </button>
          </div>
        </div>
      </div>

      {/* FLOATING TOAST FEEDBACK (Shows PostgreSQL / Spotify Like Sync) */}
      {toastMessage && (
        <div className="fixed bottom-5 right-5 z-50 bg-zinc-900/95 border border-white/20 text-white px-4 py-2.5 rounded-xl shadow-2xl flex items-center gap-2.5 text-xs font-medium backdrop-blur-md animate-bounce">
          <CheckCircle2 size={15} className="text-emerald-400 flex-shrink-0" />
          <span>{toastMessage}</span>
        </div>
      )}

      {/* MAIN PREVIEW STAGE */}
      <main className="flex-1 flex items-center justify-center p-4 sm:p-6 lg:p-8">
        {viewportMode === 'mobile' ? (
          /* ==========================================
             MOBILE PWA FRAME (390px Authentic Sleeve Mockup)
             ========================================== */
          <div className="flex flex-col items-center">
            <div
              style={{ backgroundColor: activeTheme.bg }}
              className="w-[375px] sm:w-[390px] rounded-[42px] border-[8px] border-zinc-800 shadow-[0_25px_70px_rgba(0,0,0,0.75)] overflow-hidden flex flex-col justify-between transition-colors duration-500 min-h-[760px] max-h-[820px]"
            >
              {/* Scrollable Screen Body */}
              <div className="flex-1 overflow-y-auto px-5 pt-5 pb-6 no-scrollbar">
                {renderAppHeader()}
                {renderActiveScreenContent(false)}
              </div>

              {/* Sticky Bottom Tab Bar */}
              {renderBottomNav()}
            </div>

            <p className="text-xs text-zinc-400 mt-3 text-center">
              Tip: Click any album sleeve or track row on <strong>Today</strong> or{' '}
              <strong>Library</strong> to see the ambient sleeve color adapt live.
            </p>
          </div>
        ) : (
          /* ==========================================
             DESKTOP RESPONSIVE WEB APP MODE (Full Width)
             ========================================== */
          <div
            style={{ backgroundColor: activeTheme.bg }}
            className="w-full max-w-6xl min-h-[720px] rounded-3xl border border-white/15 shadow-2xl overflow-hidden grid grid-cols-12 transition-colors duration-500"
          >
            {/* Left Desktop Sidebar Navigation */}
            <aside className="col-span-3 lg:col-span-2 bg-black/25 border-r border-white/10 p-5 flex flex-col justify-between">
              <div className="space-y-6">
                <div className="flex items-center justify-between">
                  <span
                    style={{ fontFamily: "Georgia, 'Playfair Display', serif" }}
                    className="text-xl text-white tracking-tight"
                  >
                    Scrobbler
                  </span>
                  <span className="text-[10px] uppercase tracking-wider px-2 py-0.5 rounded bg-white/10 text-white/70">
                    PWA
                  </span>
                </div>

                <nav className="space-y-1">
                  {[
                    { id: 'today', label: 'Today', icon: Activity },
                    { id: 'library', label: 'Library', icon: Library, badge: totalLikedCount },
                    { id: 'gigs', label: 'Gigs', icon: Ticket, badge: '24' },
                    { id: 'reports', label: 'Reports', icon: BarChart2 },
                    { id: 'connect', label: 'Connect Sources', icon: Radio },
                    { id: 'following', label: 'Following', icon: Users },
                    { id: 'profile', label: 'Profile', icon: Disc },
                    { id: 'settings', label: 'Settings', icon: SettingsIcon }
                  ].map((item) => {
                    const IconComponent = item.icon;
                    const active =
                      activeScreen === item.id ||
                      (item.id === 'gigs' && activeScreen === 'gig-detail');
                    return (
                      <button
                        key={item.id}
                        type="button"
                        onClick={() => setActiveScreen(item.id)}
                        className={`w-full px-3 py-2.5 rounded-xl text-xs font-semibold flex items-center justify-between transition ${
                          active
                            ? 'bg-white text-zinc-900 shadow-sm'
                            : 'text-white/75 hover:bg-white/10 hover:text-white'
                        }`}
                      >
                        <span className="flex items-center gap-2.5">
                          <IconComponent size={16} />
                          {item.label}
                        </span>
                        {item.badge !== undefined && (
                          <span
                            className={`text-[10px] px-1.5 py-0.5 rounded-full tabular-nums ${
                              active ? 'bg-zinc-900 text-white' : 'bg-white/15 text-white/85'
                            }`}
                          >
                            {item.badge}
                          </span>
                        )}
                      </button>
                    );
                  })}
                </nav>
              </div>

              {/* Mini Now Playing Sleeve Footer in Sidebar */}
              <div className="pt-4 border-t border-white/10 space-y-2">
                <div className="flex items-center gap-2.5">
                  <div
                    style={{ backgroundColor: activeHeroTrack.tileColor }}
                    className="w-9 h-9 rounded flex-shrink-0 flex items-end p-1 text-white font-serif text-xs border border-white/15"
                  >
                    {activeHeroTrack.monogram}
                  </div>
                  <div className="min-w-0 flex-1">
                    <p className="text-xs font-semibold text-white truncate">
                      {activeHeroTrack.title}
                    </p>
                    <p className="text-[11px] text-white/65 truncate">
                      {activeHeroTrack.artist}
                    </p>
                  </div>
                  <HeartLikeButton
                    liked={activeHeroTrack.liked}
                    size={15}
                    onToggle={(e) =>
                      handleToggleLike(activeHeroTrack.id, activeHeroTrack.title, e)
                    }
                  />
                </div>
              </div>
            </aside>

            {/* Desktop Main Content Area */}
            <section className="col-span-9 lg:col-span-10 p-6 lg:p-8 overflow-y-auto">
              <div className="flex items-center justify-between pb-6 mb-6 border-b border-white/10">
                <div>
                  <span className="text-[11px] uppercase tracking-widest text-white/60">
                    B — Sleeve Desktop Layout
                  </span>
                  <h2
                    style={{ fontFamily: "Georgia, 'Playfair Display', serif" }}
                    className="text-lg text-white/90"
                  >
                    Chris Lettis (@chris) • 18,432 Total Scrobbles
                  </h2>
                </div>
                <div className="flex items-center gap-3">
                  <button
                    type="button"
                    onClick={() => {
                      setOnlyLikedFilter(true);
                      setActiveScreen('library');
                    }}
                    className="px-3.5 py-1.5 rounded-full bg-white/10 hover:bg-white/15 border border-white/15 text-xs font-semibold text-white inline-flex items-center gap-1.5 transition"
                  >
                    <Heart size={13} className="fill-rose-400 text-rose-400" />
                    {totalLikedCount} Liked Songs
                  </button>
                  {renderAppHeader()}
                </div>
              </div>

              {renderActiveScreenContent(true)}
            </section>
          </div>
        )}
      </main>

      {/* MODAL 1: LOG A GIG */}
      {showLogGigModal && (
        <div className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-[#2a0c10] border border-white/20 rounded-2xl max-w-sm w-full p-5 space-y-4 shadow-2xl">
            <h3
              style={{ fontFamily: "Georgia, 'Playfair Display', serif" }}
              className="text-xl text-white"
            >
              Log a gig
            </h3>
            <form onSubmit={handleAddGig} className="space-y-3">
              <div>
                <label className="block text-xs text-white/75 mb-1">Artist / Headliner</label>
                <input
                  type="text"
                  required
                  value={newGigArtist}
                  onChange={(e) => setNewGigArtist(e.target.value)}
                  placeholder="e.g. Fontaines D.C."
                  className="w-full px-3.5 py-2 rounded-lg bg-black/30 border border-white/20 text-sm text-white placeholder-white/40 focus:outline-none focus:border-white"
                />
              </div>
              <div>
                <label className="block text-xs text-white/75 mb-1">Venue & City</label>
                <input
                  type="text"
                  required
                  value={newGigVenue}
                  onChange={(e) => setNewGigVenue(e.target.value)}
                  placeholder="e.g. Alexandra Palace, London"
                  className="w-full px-3.5 py-2 rounded-lg bg-black/30 border border-white/20 text-sm text-white placeholder-white/40 focus:outline-none focus:border-white"
                />
              </div>
              <div className="flex justify-end gap-2 pt-2">
                <button
                  type="button"
                  onClick={() => setShowLogGigModal(false)}
                  className="px-4 py-2 rounded-full border border-white/25 text-xs font-semibold text-white"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-4 py-2 rounded-full bg-white text-[#380d11] text-xs font-bold"
                >
                  Save Gig
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* MODAL 2: COMPARE LISTENING WITH FRIENDS */}
      {showCompareModal && (
        <div className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-[#0c241b] border border-white/20 rounded-2xl max-w-md w-full p-5 space-y-4 shadow-2xl">
            <div className="flex items-center justify-between">
              <h3
                style={{ fontFamily: "Georgia, 'Playfair Display', serif" }}
                className="text-xl text-white"
              >
                Family & Friends Leaderboard
              </h3>
              <button
                type="button"
                onClick={() => setShowCompareModal(false)}
                className="text-xs text-white/70 hover:text-white"
              >
                Close
              </button>
            </div>
            <p className="text-xs text-white/75">
              Weekly scrobble comparison across your self-hosted server:
            </p>
            <div className="divide-y divide-white/15 border-t border-b border-white/15">
              {friends.map((f, i) => (
                <div key={f.id} className="py-2.5 flex items-center justify-between">
                  <div className="flex items-center gap-2.5">
                    <span className="text-xs text-white/60 w-4 tabular-nums">#{i + 1}</span>
                    <span className="text-sm font-semibold text-white">{f.name}</span>
                  </div>
                  <span className="text-xs font-bold text-emerald-300 tabular-nums">
                    {f.weeklyPlays} plays
                  </span>
                </div>
              ))}
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
